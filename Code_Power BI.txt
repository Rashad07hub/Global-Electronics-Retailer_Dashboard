29. TotalCustomers = DISTINCTCOUNT(Sales[CustomerKey])

    SAMEPERIODLASTYEAR('Date'[Date])

DIVIDE(
    [TotalSales],
    [TotalOrders]
)

DIVIDE(
    [TotalSales],
    [TotalQuantity]
)


3. Avg. Revenue PerProduct = DIVIDE(
     [TotalSales],
     [Products Sold]

     ALL(Customers[Name]),
     [TotalSales],

CALCULATE(
    [TotalOrders],
    SAMEPERIODLASTYEAR('Date'[Date])

    [Orders PY]

9. Product Profitability % = DIVIDE(
     [TotalProfit],
     [TotalSales]
)

RANKX(

    [TotalSales],
    ,
    DESC,
    DENSE

1. TotalCustomers = DISTINCTCOUNT(Sales[CustomerKey]) 
2. TotalCustomers PreviousYear = 
CALCULATE(
    [TotalCustomers],
    SAMEPERIODLASTYEAR('Date'[Date])
)
3. SalesPer Customer = DIVIDE(
     [TotalSales],
     [TotalCustomers]
)
4. Customers YoY% = DIVIDE(
    [TotalCustomers] - [TotalCustomers PreviousYear],
    [TotalCustomers PreviousYear]
)
5. Customer Rank = RANKX(
     ALL(Customers[Name]),
     [TotalSales],
     ,
     DESC
)

6. TotalQuantity = SUM(Sales[Quantity])
7. TotalOrders = DISTINCTCOUNT(Sales[Order Number]) 
8. Orders PY = 
CALCULATE(
    [TotalOrders],
    SAMEPERIODLASTYEAR('Date'[Date])
)
9. AverageOrderValue = 
DIVIDE(
    [TotalSales],
    [TotalOrders]
)
10. Orders YoY % = 
DIVIDE(
    [TotalOrders] - [Orders PY],
    [Orders PY]
)

11. Products Sold = DISTINCTCOUNT(
     Sales[ProductKey])
12. ProductSold PY = CALCULATE(
    [Products Sold],
    SAMEPERIODLASTYEAR('Date'[Date])
)

13. SalesAmount = Sales[Quantity] * RELATED(Products[Unit Price USD])
14. TotalSales = SUM(Sales[SalesAmount])
15. Sales PreviousYear = CALCULATE(
    [TotalSales],
    SAMEPERIODLASTYEAR('Date'[Date])
)
16. AverageSellingPrice = 
DIVIDE(
    [TotalSales],
    [TotalQuantity]
)
17. Avg. Revenue PerProduct = DIVIDE(
     [TotalSales],
     [Products Sold]
)
18. Sales YoY % = 
DIVIDE(
    [TotalSales] - [Sales PreviousYear],
    [Sales PreviousYear]
)
19. YTD Sales = TOTALYTD(
     [TotalSales],
     'Date'[Date]
)

20. CostAmount = Sales[Quantity] * RELATED(Products[Unit Cost USD])
21. TotalCost = SUM(Sales[CostAmount]) 

22. TotalProfit = [TotalSales] - [TotalCost]
23. Profit PreviousYear = 
CALCULATE(
    [TotalProfit],
    SAMEPERIODLASTYEAR('Date'[Date])
)
24. Product Profitability % = DIVIDE(
     [TotalProfit],
     [TotalSales]
)
25. Profit Contribution % = 
DIVIDE(
    [TotalProfit],
    CALCULATE(
        [TotalProfit],
        ALL(Products[Brand])
    ),
    0
)
26.  YTD Profit = TOTALYTD(
    [TotalProfit],
    'Date'[Date]
)
27. Profit YoY % = DIVIDE(
    [TotalProfit] - [Profit PreviousYear],
    [Profit PreviousYear]
)
28. ProfitContribution Country % = 
DIVIDE(
    [TotalProfit],
    CALCULATE(
        [TotalProfit],
        ALL(Customers[Country])
    ),
    0
)
29. ProfitMargin = DIVIDE(
    [TotalProfit],
    [TotalSales]
)
30. ProfitMargin PreviousYear = 
CALCULATE(
    [ProfitMargin],
    SAMEPERIODLASTYEAR('Date'[Date])
)
31. ProfitMargin PY = 
CALCULATE(
    [ProfitMargin],
    SAMEPERIODLASTYEAR('Date'[Date])
)

32. ProfitMargin YoY % = DIVIDE(
    [ProfitMargin] - [ProfitMargin PY],
    [ProfitMargin PY]
)

33. Product Rank = 
RANKX(
    ALLSELECTED(Products[Product Name]),
    [TotalSales],
    ,
    DESC,
    DENSE
)
34. Selected Top N = 
SELECTEDVALUE(
    'Top N Selection'[Top N],
    10
)
35. Show Top N Product = 
VAR N = 'Sales'[Selected Top N]
VAR ProductRank = [Product Rank]
RETURN
IF(
    ProductRank <= N,
    1,
    0
)

36. Top N Product Title = 
"Top "
    & [Selected Top N]
    & " Products by Sales"

37. YoY Arrow = 
IF(
    [Sales YoY %] > 0,
    "▲",
    IF(
        [Sales YoY %] < 0,
        "▼",
        "—"
    )
)

38. YoY Color = 
IF(
    [Sales YoY %] > 0,
    "#008000",
    IF(
        [Sales YoY %] < 0,
        "#C00000",
        "#808080"
    )
)

39. YoY Display = 
VAR Growth = [Sales YoY %]
VAR Arrow =
    SWITCH(
        TRUE(),
        Growth > 0, "▲ ",
        Growth < 0, "▼ ",
        "— "
    )
RETURN
    Arrow & " " & FORMAT(Growth, "0.0%") & " YoY"
